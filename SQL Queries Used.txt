/*
================================================================================
SALES INCENTIVE COMPENSATION MANAGEMENT (ICM) — SQL PROJECT QUERIES
Fictional organisation: ABC Financial Services
Database: SQLite | Tool used: DB Browser for SQLite

PURPOSE
-------
This script documents the SQL work completed for the independent ICM learning
project, from table creation and CSV import instructions through data cleaning,
table population, incentive calculations, validation and final reporting.

IMPORTANT PROJECT BOUNDARIES
----------------------------
1. All employee, product, target and transaction data are fictional.
2. Incentive-plan figures are illustrative assumptions, not industry benchmarks.
3. This is an independent learning simulation, NOT a Varicent implementation,
   production ICM system or live payroll process.
4. The Excel workbook and SQLite database independently implement the same
   core business rules so that their outputs can be reconciled.
5. The two incentive components have SEPARATE caps:
      Achievement incentive cap   = ₹30,000
      Profitability incentive cap = ₹5,000
      Maximum combined payout     = ₹35,000 per employee per month.
6. The profitability incentive is independent of the 80% achievement threshold.
7. Employee joining and exit dates are treated as inclusive dates in this
   simulation. Employee master dates were imported as DD-MM-YYYY text;
   transaction and incentive month dates are stored as YYYY-MM-DD text.

NOTE:
-CSV IMPORTS WERE PERFORMED THROUGH DB BROWSER'S "IMPORT -> TABLE FROM CSV FILE"
FUNCTIONALITY. SQL itself does not browse to or import a local file path here.
For each import:
  a. Open the correct database.
  b. Choose File -> Import -> Table from CSV file.
  c. Select the CSV exported from the corresponding Excel sheet.
  d. Use the staging-table name and source headers documented below.
  e. Confirm the first row contains column names.
  f. Import the data, then run the corresponding INSERT ... SELECT below.

SOURCE SHEETS / CSV IMPORTS
------------------------------------
Excel workbook sheets exported as CSV:
  Employee_Master       -> Employee_Master_Staging  (100 rows)
  Product_Master        -> Product_Master_Staging   (8 rows)
  Sales_Transactions    -> Sales_Transactions_Staging (1,500 rows)
  Target_Master         -> Target_Master_Staging    (300 rows)

The staging tables are temporary import landing areas in the project database.
The INSERT ... SELECT statements below convert and transfer the imported
records into the final structured tables.

Expected source column headers:
  Employee_Master:
    Employee_ID, Employee Name, Region, Role, Joining Date,
    Employment End Date, Employment Status, Manager ID

  Product_Master:

  Sales_Transactions:
    Transaction ID, Employee ID, Product, Sales Value (INR),
    Transaction Date, Transaction Status

  Target_Master:
    Target ID, Employee ID, Incentive Month, Sales Target,
    Eligible Service Days, Calender Days, Adjusted Sales Target

NOTE:

/*==============================================================================
SECTION 1 — CREATION OF FINAL DATABASE TABLES
==============================================================================*/

-- 1A. Employee master:
-- One row per employee. Employee_ID is the primary key.
CREATE TABLE IF NOT EXISTS Employee_Master (
    Employee_ID TEXT PRIMARY KEY,
    Employee_Name TEXT,
    Region TEXT,
    Role TEXT,
    Joining_Date TEXT,
    Employment_End_Date TEXT,
    Employment_Status TEXT,
    Manager_ID TEXT
);

-- 1B. Product master:
-- One row per product. Contribution_Margin is stored as a decimal:
-- 15% is stored as 0.15.
CREATE TABLE IF NOT EXISTS Product_Master (
    Product_ID TEXT PRIMARY KEY,
    Product_Name TEXT NOT NULL,
    Standard_Selling_Price REAL NOT NULL,
    Contribution_Margin REAL NOT NULL,
    Profitability_Eligibility TEXT
);

-- 1C. Sales transactions:
-- One row per transaction. Employee_ID and Product_ID reference master tables.
CREATE TABLE IF NOT EXISTS Sales_Transactions (
    Transaction_ID TEXT PRIMARY KEY,
    Employee_ID TEXT NOT NULL,
    Product_ID TEXT NOT NULL,
    Sales_Value REAL NOT NULL,
    Transaction_Date TEXT NOT NULL,
    Transaction_Status TEXT NOT NULL,
    FOREIGN KEY (Employee_ID) REFERENCES Employee_Master(Employee_ID),
    FOREIGN KEY (Product_ID) REFERENCES Product_Master(Product_ID)
);

-- 1D. Monthly target master:
-- One row per employee per incentive month.
CREATE TABLE IF NOT EXISTS Target_Master (
    Target_ID TEXT PRIMARY KEY,
    Employee_ID TEXT NOT NULL,
    Incentive_Month TEXT NOT NULL,
    Sales_Target REAL NOT NULL,
    Eligible_Service_Days INTEGER,
    Calendar_Days INTEGER,
    Adjusted_Sales_Target REAL,
    FOREIGN KEY (Employee_ID) REFERENCES Employee_Master(Employee_ID),
    UNIQUE (Employee_ID, Incentive_Month)
);

/*==============================================================================
SECTION 3 — TRANSFER AND CLEANING OF EMPLOYEE MASTER DATA
Source: Employee_Master_Staging, imported from the Excel Employee_Master sheet.
The source dates are DD-MM-YYYY text. They are retained in that format in the
final Employee_Master table and standardised to YYYY-MM-DD inside date queries.
Blank Employment End Date values are converted to NULL.
==============================================================================*/

INSERT INTO Employee_Master (
    Employee_ID,
    Employee_Name,
    Region,
    Role,
    Joining_Date,
    Employment_End_Date,
    Employment_Status,
    Manager_ID
)
SELECT
    TRIM(Employee_ID),
    TRIM("Employee Name"),
    TRIM(Region),
    TRIM(Role),
    TRIM("Joining Date"),
    CASE
        WHEN "Employment End Date" IS NULL
             OR TRIM("Employment End Date") = ''
        THEN NULL
        ELSE TRIM("Employment End Date")
    END,
    TRIM("Employment Status"),
    NULLIF(TRIM("Manager ID"), '')
FROM Employee_Master_Staging;


/*==============================================================================
SECTION 4 — TRANSFER AND CLEANING OF PRODUCT MASTER DATA
Source: Product_Master_Staging, imported from the Excel Product_Master sheet.
The source price may contain ₹, commas and spaces. These are removed before
CAST converts the value to a numeric amount.
The margin source may contain a % sign. It is converted to a decimal:
  15% -> 0.15
==============================================================================*/

INSERT INTO Product_Master (
    Product_ID,
    Product_Name,
    Standard_Selling_Price,
    Contribution_Margin,
    Profitability_Eligibility
)
SELECT
    TRIM(Product_ID),
    TRIM(Product_Name),
    CAST(
        REPLACE(
            REPLACE(
                REPLACE(Standard_Selling_Price, '₹', ''),
            ',', ''),
        ' ', '') AS REAL
    ),
    CAST(
        REPLACE(
            REPLACE("Contribution_Margin_%", '%', ''),
        ' ', '') AS REAL
    ) / 100.0,
    CASE
        WHEN CAST(
            REPLACE(
                REPLACE("Contribution_Margin_%", '%', ''),
            ' ', '') AS REAL
        ) / 100.0 >= 0.15
        THEN 'Yes'
        ELSE 'No'
    END
FROM Product_Master_Staging;


/*==============================================================================
SECTION 5 — TRANSFER AND CLEANING OF SALES TRANSACTIONS
Source: Sales_Transactions_Staging, imported from the Excel
Sales_Transactions sheet. 
The source contains Product_Name, while the final table stores Product_ID.
JOIN to Product_Master converts the product name to its product key.
Sales values are cleaned of ₹, commas and spaces.
DD-MM-YYYY transaction dates are converted to YYYY-MM-DD.
==============================================================================*/

INSERT INTO Sales_Transactions (
    Transaction_ID,
    Employee_ID,
    Product_ID,
    Sales_Value,
    Transaction_Date,
    Transaction_Status
)
SELECT
    TRIM(S."Transaction ID"),
    TRIM(S."Employee ID"),
    P.Product_ID,
    CAST(
        REPLACE(
            REPLACE(
                REPLACE(S."Sales Value (INR)", '₹', ''),
            ',', ''),
        ' ', '') AS REAL
    ),
    SUBSTR(TRIM(S."Transaction Date"), 7, 4) || '-' ||
    SUBSTR(TRIM(S."Transaction Date"), 4, 2) || '-' ||
    SUBSTR(TRIM(S."Transaction Date"), 1, 2),
    TRIM(S."Transaction Status")
FROM Sales_Transactions_Staging AS S
JOIN Product_Master AS P
    ON TRIM(S.Product) = P.Product_Name;


/*==============================================================================
SECTION 6 — TRANSFER AND CLEANING OF MONTHLY TARGET DATA
Source: Target_Master_Staging, imported from the Excel Target_Master sheet.
Incentive Month is imported in a format such as Sep-26 and converted to
YYYY-MM-01, for example 2026-09-01.
Currency values are cleaned before conversion to REAL.
"Calender Days" is the original source header and is mapped to Calendar_Days.
==============================================================================*/

INSERT INTO Target_Master (
    Target_ID,
    Employee_ID,
    Incentive_Month,
    Sales_Target,
    Eligible_Service_Days,
    Calendar_Days,
    Adjusted_Sales_Target
)
SELECT
    TRIM("Target ID"),
    TRIM("Employee ID"),
    '20' || SUBSTR(TRIM("Incentive Month"), 5, 2) || '-' ||
    CASE SUBSTR(TRIM("Incentive Month"), 1, 3)
        WHEN 'Jan' THEN '01'
        WHEN 'Feb' THEN '02'
        WHEN 'Mar' THEN '03'
        WHEN 'Apr' THEN '04'
        WHEN 'May' THEN '05'
        WHEN 'Jun' THEN '06'
        WHEN 'Jul' THEN '07'
        WHEN 'Aug' THEN '08'
        WHEN 'Sep' THEN '09'
        WHEN 'Oct' THEN '10'
        WHEN 'Nov' THEN '11'
        WHEN 'Dec' THEN '12'
    END || '-01',
    CAST(
        REPLACE(
            REPLACE(
                REPLACE("Sales Target", '₹', ''),
            ',', ''),
        ' ', '') AS REAL
    ),
    CAST("Eligible Service Days" AS INTEGER),
    CAST("Calender Days" AS INTEGER),
    CAST(
        REPLACE(
            REPLACE(
                REPLACE("Adjusted Sales Target", '₹', ''),
            ',', ''),
        ' ', '') AS REAL
    )
FROM Target_Master_Staging;

/*==============================================================================
SECTION 12 — CREATE THE FINAL INCENTIVE CALCULATION VIEW
This is the central SQL calculation.

The view starts from Target_Master so that all 300 employee-month target rows
remain present, including employees with no eligible sales.

Business rules applied:
  1. Only Completed transactions are considered for eligible sales.
  2. Transaction must fall within the incentive month.
  3. Transaction must be on/after joining date and on/before exit date.
  4. Achievement = eligible net sales / adjusted target.
  5. Achievement below 80% -> no achievement incentive.
  6. 80% to 100% -> linear ramp from ₹0 to ₹22,500.
  7. Above 100% -> 1.5× accelerator on achievement above 100%.
  8. Achievement incentive is capped separately at ₹30,000.
  9. Qualifying product margin >= 15%.
 10. Profitability incentive = 0.5% of qualifying net sales, capped at ₹5,000.
 11. Final payout = capped achievement incentive + capped profitability
     incentive. Maximum combined payout = ₹35,000.

Important:
- A LEFT JOIN from Target_Master preserves zero-sales employee-months.
- Transaction filters are placed in the JOIN condition so that the LEFT JOIN
  does not accidentally remove employees with no matching eligible sales.
==============================================================================*/

DROP VIEW IF EXISTS Incentive_Calculation;

CREATE VIEW Incentive_Calculation AS
WITH
/* CTE 1: Convert employee joining and exit dates into ISO format. */
Employee_Dates AS (
    SELECT
        Employee_ID,
        CASE
            WHEN Joining_Date IS NULL OR TRIM(Joining_Date) = ''
            THEN NULL
            ELSE
                SUBSTR(Joining_Date, 7, 4) || '-' ||
                SUBSTR(Joining_Date, 4, 2) || '-' ||
                SUBSTR(Joining_Date, 1, 2)
        END AS Standard_Joining_Date,
        CASE
            WHEN Employment_End_Date IS NULL
                 OR TRIM(Employment_End_Date) = ''
            THEN NULL
            ELSE
                SUBSTR(Employment_End_Date, 7, 4) || '-' ||
                SUBSTR(Employment_End_Date, 4, 2) || '-' ||
                SUBSTR(Employment_End_Date, 1, 2)
        END AS Standard_Exit_Date
    FROM Employee_Master
),

/* CTE 2: Aggregate eligible completed sales for every employee-month target.
   Starting from Target_Master retains rows even when there are no eligible
   transactions. */
Sales_Achievement AS (
    SELECT
        T.Employee_ID,
        T.Incentive_Month,
        T.Adjusted_Sales_Target,

        COALESCE(SUM(S.Sales_Value), 0) AS Eligible_Net_Sales

    FROM Target_Master AS T

    LEFT JOIN Employee_Dates AS E
        ON T.Employee_ID = E.Employee_ID

    LEFT JOIN Sales_Transactions AS S
        ON T.Employee_ID = S.Employee_ID

        -- Include transactions within the incentive month.
        AND S.Transaction_Date >= T.Incentive_Month
        AND S.Transaction_Date < DATE(T.Incentive_Month, '+1 month')

        -- Only completed transactions are eligible.
        AND S.Transaction_Status = 'Completed'

        -- Exclude transactions before the employee joined.
        AND S.Transaction_Date >= E.Standard_Joining_Date

        -- Exclude transactions after the employee exited.
        AND (
            E.Standard_Exit_Date IS NULL
            OR S.Transaction_Date <= E.Standard_Exit_Date
        )

    GROUP BY
        T.Employee_ID,
        T.Incentive_Month,
        T.Adjusted_Sales_Target
),

/* CTE 3: Calculate achievement percentage and achievement eligibility.
   NULLIF prevents division by zero when adjusted target is zero.
   COALESCE converts the resulting NULL achievement to zero. */
Eligibility AS (
    SELECT
        Employee_ID,
        Incentive_Month,
        Adjusted_Sales_Target,
        Eligible_Net_Sales,

        COALESCE(
            Eligible_Net_Sales / NULLIF(Adjusted_Sales_Target, 0),
            0
        ) AS Achievement_Percentage,

        CASE
            WHEN Adjusted_Sales_Target = 0
                THEN 'Not Eligible - Zero Target'
            WHEN COALESCE(
                Eligible_Net_Sales / NULLIF(Adjusted_Sales_Target, 0),
                0
            ) < 0.80
                THEN 'Not Eligible - Below Threshold'
            ELSE 'Eligible'
        END AS Eligibility_Status

    FROM Sales_Achievement
),

/* CTE 4: Calculate achievement incentive before applying its component cap. */
Achievement_Payout AS (
    SELECT
        Employee_ID,
        Incentive_Month,
        Adjusted_Sales_Target,
        Eligible_Net_Sales,
        Achievement_Percentage,
        Eligibility_Status,

        CASE
            WHEN Eligibility_Status <> 'Eligible'
                THEN 0

            -- Linear payout ramp from 80% achievement to 100%.
            WHEN Achievement_Percentage <= 1.00
                THEN 22500.0 *
                     (Achievement_Percentage - 0.80) /
                     (1.00 - 0.80)

            -- Above 100%, apply the 1.5x accelerator only to the excess.
            ELSE
                22500.0 +
                (
                    (Achievement_Percentage - 1.00)
                    * 22500.0
                    * 1.5
                )
        END AS Achievement_Incentive_Before_Cap

    FROM Eligibility
),

/* CTE 5: Calculate qualifying net sales.
   A transaction qualifies only when:
     - it is Completed;
     - it falls within the employee's employment period;
     - it falls within the incentive month; and
     - its product contribution margin is at least 15%.
   This component is independent of the 80% achievement threshold. */
Profitability_Payout AS (
    SELECT
        T.Employee_ID,
        T.Incentive_Month,

        COALESCE(SUM(
            CASE
                WHEN P.Contribution_Margin >= 0.15
                THEN S.Sales_Value
                ELSE 0
            END
        ), 0) AS Qualifying_Net_Sales

    FROM Target_Master AS T

    LEFT JOIN Employee_Dates AS E
        ON T.Employee_ID = E.Employee_ID

    LEFT JOIN Sales_Transactions AS S
        ON T.Employee_ID = S.Employee_ID
        AND S.Transaction_Date >= T.Incentive_Month
        AND S.Transaction_Date < DATE(T.Incentive_Month, '+1 month')
        AND S.Transaction_Status = 'Completed'
        AND S.Transaction_Date >= E.Standard_Joining_Date
        AND (
            E.Standard_Exit_Date IS NULL
            OR S.Transaction_Date <= E.Standard_Exit_Date
        )

    LEFT JOIN Product_Master AS P
        ON S.Product_ID = P.Product_ID

    GROUP BY
        T.Employee_ID,
        T.Incentive_Month
),

/* CTE 6: Combine both independently calculated incentive components and
   apply each component's separate cap. */
Final_Payout AS (
    SELECT
        A.Employee_ID,
        A.Incentive_Month,
        A.Adjusted_Sales_Target,
        A.Eligible_Net_Sales,
        A.Achievement_Percentage,
        A.Eligibility_Status,
        A.Achievement_Incentive_Before_Cap,

        MIN(
            MAX(A.Achievement_Incentive_Before_Cap, 0),
            30000.0
        ) AS Achievement_Incentive_After_Cap,

        P.Qualifying_Net_Sales,

        MIN(
            P.Qualifying_Net_Sales * 0.005,
            5000.0
        ) AS Profitability_Incentive

    FROM Achievement_Payout AS A

    LEFT JOIN Profitability_Payout AS P
        ON A.Employee_ID = P.Employee_ID
        AND A.Incentive_Month = P.Incentive_Month
)

-- Final output: one row per employee per incentive month.
SELECT
    Employee_ID,
    Incentive_Month,
    Adjusted_Sales_Target,
    Eligible_Net_Sales,
    Achievement_Percentage,
    Eligibility_Status,
    Achievement_Incentive_Before_Cap,
    Achievement_Incentive_After_Cap,
    Qualifying_Net_Sales,
    Profitability_Incentive,

    -- Add the two separately capped components.
    Achievement_Incentive_After_Cap
        + Profitability_Incentive AS Total_Incentive

FROM Final_Payout;


/*==============================================================================
SECTION 13 — VIEW THE FINAL EMPLOYEE-MONTH INCENTIVE OUTPUT

SELECT *
FROM Incentive_Calculation
ORDER BY Incentive_Month, Employee_ID;

/*==============================================================================

-- 15A. Sum of achievement and profitability incentive components and final payout.
SELECT
    SUM(Achievement_Incentive_After_Cap) AS Total_Achievement_Incentive,
    SUM(Profitability_Incentive) AS Total_Profitability_Incentive,
    SUM(Total_Incentive) AS Total_Incentive_Payout
FROM Incentive_Calculation;

-- 15B. Check the maximum employee-month payout.
SELECT
    MAX(Total_Incentive) AS Maximum_Employee_Month_Payout
FROM Incentive_Calculation;

-- 15C. Check whether any payout exceeds the maximum combined cap.
SELECT
    COUNT(*) AS Payouts_Above_Combined_Cap
FROM Incentive_Calculation
WHERE Total_Incentive > 35000;

-- 15D. Check component-to-total reconciliation.
SELECT
    SUM(Achievement_Incentive_After_Cap)
      + SUM(Profitability_Incentive)
      - SUM(Total_Incentive) AS Reconciliation_Difference
FROM Incentive_Calculation;

-- 15E. Verify total qualifying sales against the qualifying source transactions.
-- Source criteria: Completed, within employment, product margin >= 15%.
SELECT
    SUM(Qualifying_Net_Sales) AS Qualifying_Sales_From_View
FROM Incentive_Calculation;

-- 15F. Inspect the highest payout employee-month records.
SELECT
    Employee_ID,
    Incentive_Month,
    Achievement_Percentage,
    Achievement_Incentive_After_Cap,
    Qualifying_Net_Sales,
    Profitability_Incentive,
    Total_Incentive
FROM Incentive_Calculation
ORDER BY Total_Incentive DESC
LIMIT 20;


/*==============================================================================
SECTION 16 — CROSS-CHECK TOTAL ELIGIBLE SALES
==============================================================================*/

WITH Employee_Dates AS (
    SELECT
        Employee_ID,
        SUBSTR(Joining_Date, 7, 4) || '-' ||
        SUBSTR(Joining_Date, 4, 2) || '-' ||
        SUBSTR(Joining_Date, 1, 2) AS Standard_Joining_Date,
        CASE
            WHEN Employment_End_Date IS NULL
                 OR TRIM(Employment_End_Date) = ''
            THEN NULL
            ELSE
                SUBSTR(Employment_End_Date, 7, 4) || '-' ||
                SUBSTR(Employment_End_Date, 4, 2) || '-' ||
                SUBSTR(Employment_End_Date, 1, 2)
        END AS Standard_Exit_Date
    FROM Employee_Master
)
SELECT
    SUM(S.Sales_Value) AS Total_Eligible_Net_Sales
FROM Sales_Transactions AS S
JOIN Employee_Dates AS E
    ON S.Employee_ID = E.Employee_ID
WHERE S.Transaction_Status = 'Completed'
  AND S.Transaction_Date >= E.Standard_Joining_Date
  AND (
      E.Standard_Exit_Date IS NULL
      OR S.Transaction_Date <= E.Standard_Exit_Date
  );


/*==============================================================================
SECTION 17 — FINAL PROJECT TABLE AND VIEW INVENTORY
==============================================================================*/

SELECT
    type AS Object_Type,
    name AS Object_Name
FROM sqlite_master
WHERE type IN ('table', 'view')
ORDER BY
    CASE type WHEN 'table' THEN 1 ELSE 2 END,
    name;
